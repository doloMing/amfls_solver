function(add_amfls_test name)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_link_libraries(${name} PRIVATE amfls_core)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
